 RETURNS TABLE(tipo text, nome text, email text, identificador text)
 LANGUAGE plpgsql
 SET search_path TO 'public'
AS $function$
BEGIN
  RETURN QUERY
  -- Busca pré-cadastros de Usuários/Alunos pendentes
  SELECT 
    'usuario'::text AS tipo,
    u.nome,
    u.email,
    u.matricula AS identificador
  FROM public.usuarios u
  WHERE u.uuid IS NULL

  UNION ALL

  -- Busca pré-cadastros de Administradores pendentes
  SELECT 
    'administrador'::text AS tipo,
    a.nome,
    a.email,
    a.codigo AS identificador
  FROM public.administradores a
  WHERE a.uuid IS NULL;
END;
$function$

public,decrementar_equipamento_reserva,,trigger,"CREATE OR REPLACE FUNCTION public.decrementar_equipamento_reserva()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  -- Tenta decrementar a quantidade solicitada
  UPDATE public.equipamentos
  SET quantidade = quantidade - NEW.quantidade
  WHERE id = NEW.id_equipamento AND quantidade >= NEW.quantidade;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Quantidade de equipamentos indisponível no momento.';
  END IF;

  RETURN NEW;
END;
$function$

public,incrementar_equipamento_devolucao,,trigger,"CREATE OR REPLACE FUNCTION public.incrementar_equipamento_devolucao()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  -- Recompõe o estoque se a reserva for CANCELADA ou devolvida
  IF (
    (NEW.status = 'cancelada' AND OLD.status != 'cancelada') OR
    (NEW.status_devolucao = 'devolvido' AND OLD.status_devolucao != 'devolvido')
  ) THEN
    UPDATE public.equipamentos
    SET quantidade = quantidade + OLD.quantidade
    WHERE id = NEW.id_equipamento;
  END IF;

  RETURN NEW;
END;
$function$

 RETURNS TABLE(tipo_perfil text, id bigint, nome text, email text)
 LANGUAGE plpgsql
 SET search_path TO 'public'
AS $function$
BEGIN
  -- Impede consulta de perfis de terceiros
  IF auth.uid() IS NULL OR auth.uid() <> p_uuid THEN
    RAISE EXCEPTION 'Acesso negado: você só pode consultar seu próprio perfil.';
  END IF;

  -- Busca na tabela administradores
  RETURN QUERY
  SELECT 'admin'::text, a.id_adm, a.nome, a.email
  FROM public.administradores a
  WHERE a.uuid = p_uuid;

  IF FOUND THEN
    RETURN;
  END IF;

  -- Busca na tabela usuarios
  RETURN QUERY
  SELECT 'aluno'::text, u.id_usuario, u.nome, u.email
  FROM public.usuarios u
  WHERE u.uuid = p_uuid;

  RETURN;
END;
$function$

public,trg_validar_e_ajustar_manutencao_equipamento,,trigger,"CREATE OR REPLACE FUNCTION public.trg_validar_e_ajustar_manutencao_equipamento()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  -- Garante valor 0 para quantidade_manutencao caso seja NULL
  NEW.quantidade_manutencao := COALESCE(NEW.quantidade_manutencao, 0);

  -- Regra Automática de Status:
  -- Se o estoque disponível zerar e houver itens em manutenção, força o status para 'manutencao'
  IF NEW.quantidade = 0 AND NEW.quantidade_manutencao > 0 THEN
    NEW.status := 'manutencao';
  -- Se o status era 'manutencao' mas voltou a ter estoque livre, força o status para 'livre'
  ELSIF NEW.status = 'manutencao' AND NEW.quantidade > 0 THEN
    NEW.status := 'livre';
  END IF;

  RETURN NEW;
END;
$function$
"
