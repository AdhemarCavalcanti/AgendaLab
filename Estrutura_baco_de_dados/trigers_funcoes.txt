app_private,handle_user_activation,"CREATE OR REPLACE FUNCTION app_private.handle_user_activation()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
declare
  v_matricula text;
  v_codigo text;
begin
  -- Extrai os dados do 'options.data' passados durante o signUp
  v_matricula := new.raw_user_meta_data->>'matricula';
  v_codigo := new.raw_user_meta_data->>'codigo';

  -- Se foi fornecida matrícula, vincula o UUID no registro de usuário
  if v_matricula is not null and v_matricula <> '' then
    update public.usuarios
    set uuid = new.id
    where lower(trim(email)) = lower(trim(new.email))
      and trim(matricula) = trim(v_matricula)
      and uuid is null;
  end if;

  -- Se foi fornecido código, vincula o UUID no registro de administrador
  if v_codigo is not null and v_codigo <> '' then
    update public.administradores
    set uuid = new.id
    where lower(trim(email)) = lower(trim(new.email))
      and trim(codigo) = trim(v_codigo)
      and uuid is null;
  end if;

  return new;
end;
$function$
