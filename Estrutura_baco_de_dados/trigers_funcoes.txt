public,admin_listar_pre_cadastros,"CREATE OR REPLACE FUNCTION public.admin_listar_pre_cadastros()
 RETURNS TABLE(tipo text, nome text, email text, identificador text)
 LANGUAGE plpgsql
 SET search_path TO 'public'
AS $function$
BEGIN
  RETURN QUERY
  -- Busca pré-cadastros de Usuários/Alunos pendentes
  SELECT 
    'usuario'::text AS tipo,
    u.nome,
    u.email,
    u.matricula AS identificador
  FROM public.usuarios u
  WHERE u.uuid IS NULL

  UNION ALL

  -- Busca pré-cadastros de Administradores pendentes
  SELECT 
    'administrador'::text AS tipo,
    a.nome,
    a.email,
    a.codigo AS identificador
  FROM public.administradores a
  WHERE a.uuid IS NULL;
END;
$function$

public,decrementar_equipamento_reserva,"CREATE OR REPLACE FUNCTION public.decrementar_equipamento_reserva()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  UPDATE public.equipamentos
  SET quantidade = quantidade - 1
  WHERE id = NEW.id_equipamento AND quantidade > 0;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Equipamento indisponível no momento.';
  END IF;

  RETURN NEW;
END;
$function$

public,incrementar_equipamento_devolucao,"CREATE OR REPLACE FUNCTION public.incrementar_equipamento_devolucao()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  -- Incrementa o estoque quando o status muda para 'cancelada'
  IF (NEW.status = 'cancelada' AND OLD.status != 'cancelada') THEN
    UPDATE public.equipamentos
    SET quantidade = quantidade + 1
    WHERE id = NEW.id_equipamento;
  END IF;

  RETURN NEW;
END;
$function$

public,obter_perfil_usuario,"CREATE OR REPLACE FUNCTION public.obter_perfil_usuario(p_uuid uuid)
 RETURNS TABLE(tipo_perfil text, id bigint, nome text, email text)
 LANGUAGE plpgsql
 SET search_path TO 'public'
AS $function$
BEGIN
  -- Impede consulta de perfis de terceiros
  IF auth.uid() IS NULL OR auth.uid() <> p_uuid THEN
    RAISE EXCEPTION 'Acesso negado: você só pode consultar seu próprio perfil.';
  END IF;

  -- Busca na tabela administradores
  RETURN QUERY
  SELECT 'admin'::text, a.id_adm, a.nome, a.email
  FROM public.administradores a
  WHERE a.uuid = p_uuid;

  IF FOUND THEN
    RETURN;
  END IF;

  -- Busca na tabela usuarios
  RETURN QUERY
  SELECT 'aluno'::text, u.id_usuario, u.nome, u.email
  FROM public.usuarios u
  WHERE u.uuid = p_uuid;

  RETURN;
END;
$function$
